--[[
    STANDALONE ROBLOX PLACE ARCHIVER
    Output:
    workspace/place_archives/<PlaceName>_<PlaceId>/place.rbxlx
]]

-- CONFIG
local ROOT_FOLDER = "place_archives"
local DECOMPILE = false -- true = include decompiled script sources (much slower)

local RepoURL = "https://raw.githubusercontent.com/rbxlx/SynSaveInstance/main/"
local SSI = "saveinstance"

local MarketplaceService = game:GetService("MarketplaceService")

-- Get experience name
local placeName = "Place"
pcall(function()
    local info = MarketplaceService:GetProductInfo(game.PlaceId)
    if info and info.Name then
        placeName = info.Name
    end
end)

local function safeName(str)
    return (tostring(str):gsub("[^%w%-%._]", "_")):sub(1, 100)
end

local function ensureFolder(path)
    local current = ""
    for part in path:gmatch("[^/]+") do
        current = current == "" and part or current .. "/" .. part
        local exists = false
        if isfolder then
            pcall(function()
                exists = isfolder(current)
            end)
        end
        if not exists then
            pcall(function()
                makefolder(current)
            end)
        end
    end
end

local folder = ROOT_FOLDER .. "/" .. safeName(placeName) .. "_" .. tostring(game.PlaceId)
local filePath = folder .. "/place.rbxlx"

ensureFolder(folder)
print("[Archive] Output folder:", folder, "| exists:", isfolder and isfolder(folder))

print("[Archive] Loading Universal Syn SaveInstance...")

local okDl, source = pcall(game.HttpGet, game, RepoURL .. SSI .. ".luau", true)
print("[Archive] Download:", okDl, okDl and #source .. " chars" or source)
if not okDl then return end

local synsaveinstance = loadstring(source, SSI)()

print("[Archive] Loader returned:", typeof(synsaveinstance))

if typeof(synsaveinstance) ~= "function" then
    warn("[Archive] Loader did not return a function.")
    return
end

local options = {
    FilePath = filePath,
    Binary = false, -- always save as XML (.rbxlx)
    SafeMode = false,
    ShowStatus = true,
    ReadMe = false,

    -- Write the file in one writefile call instead of appendfile pieces.
    AlternativeWritefile = false,

    Decompile = DECOMPILE,
}

print("[Archive] Starting save...")
local success, result = pcall(function()
    return synsaveinstance(options)
end)

print("[Archive] Save call finished.")
print("[Archive] Success:", success)
print("[Archive] Result:", tostring(result))
print("[Archive] File exists:", isfile(filePath), "->", filePath)
