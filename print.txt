-- Tests which file types Solara's writefile accepts.
if not isfolder("place_archives") then
    makefolder("place_archives")
end

for _, ext in ipairs({ "txt", "rbxl", "rbxlx" }) do
    local path = "place_archives/ext_test." .. ext
    local ok, err = pcall(writefile, path, "hello")
    print(ext, "write ok:", ok, "| error:", tostring(err), "| exists:", isfile(path))
end
